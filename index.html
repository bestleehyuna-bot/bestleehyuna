💡 Supabase 테이블 생성 SQL 쿼리 보기 펼치기

Supabase 대시보드 ➔ SQL Editor ➔ New Query에 아래 SQL을 붙여넣고 [Run]을 누르세요. (supabase_schema.sql 파일 제공됨)

CREATE TABLE public.activity_logs (
    id TEXT PRIMARY KEY,
    date DATE NOT NULL DEFAULT CURRENT_DATE,
    program_type TEXT NOT NULL DEFAULT '공익활동형',
    group_name TEXT NOT NULL,
    name TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT '담당자 확인 중',
    hours NUMERIC NOT NULL DEFAULT 3,
    time_range TEXT DEFAULT '09:00 ~ 12:00',
    activity_items JSONB DEFAULT '[]'::jsonb,
    health_condition TEXT DEFAULT '양호',
    note TEXT DEFAULT '',
    reviewer TEXT DEFAULT '',
    reviewed_at TEXT DEFAULT '',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE public.activity_logs ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public on activity_logs" ON public.activity_logs FOR ALL USING (true);
ALTER PUBLICATION supabase_realtime ADD TABLE public.activity_logs;
